29def _flag(name: str, default: bool) -> bool: 

30 """Читає булеву змінну середовища: 1/true/yes/on — увімкнено.""" 

31 raw = os.environ.get(name) 

32 if raw is None: 

33 return default 

34 return raw.strip().lower() in {"1", "true", "yes", "on"} 

35

37@dataclass(frozen=True) 

38class Settings: 

39 """Налаштування процесу API.""" 

40 

41 # Записувати специфікацію у docs/api/ під час старту сервера (цикл Code-First 

42 # на локальній машині). У тестах і в майбутньому контейнері — вимкнено. 

43 export_openapi_on_startup: bool 

44 # Демо-автентифікація за токенами `demo-*` (DD-18; DD-06 — цільова схема). 

45 # Вимкнена, доки її не ввімкнули явно (V-01 у static-analysis.md). 

46 demo_auth_enabled: bool 

47 # Джерела, яким дозволено CORS (через кому). Порожньо — CORS вимкнено. 

48 cors_origins: tuple[str, ...]

51@lru_cache 

52def get_settings() -> Settings: 

53 """Повертає налаштування; кешується на весь процес.""" 

54 origins = os.environ.get("UNISLOT_CORS_ORIGINS", "") 

55 return Settings( 

56 export_openapi_on_startup=_flag("UNISLOT_EXPORT_OPENAPI", default=True), 

57 # Безпечний збій (OWASP): токени `demo-*` відкрито описані в README й OpenAPI, 

58 # серед них `demo-admin`. Сервер, запущений без змінної, їх не приймає (V-01). 

59 demo_auth_enabled=_flag("UNISLOT_DEMO_AUTH", default=False), 

60 cors_origins=tuple(o.strip() for o in origins.split(",") if o.strip()), 

61 )
